cmake_minimum_required(VERSION 3.28)
project(ArcReverse VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 WebEngineQuick Sql Test)
qt_standard_project_setup(REQUIRES 6.10)
qt_add_executable(arcreverse app/main.cpp app/workspace.cpp app/workspace.h)
qt_add_qml_module(arcreverse URI ArcReverse VERSION 1.0 QML_FILES app/Main.qml app/EngineGate.qml)
target_link_libraries(arcreverse PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick Qt6::Sql)
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(workspace_tests tests/workspace_test.cpp app/workspace.cpp app/workspace.h)
 target_include_directories(workspace_tests PRIVATE app)
 target_link_libraries(workspace_tests PRIVATE Qt6::Test Qt6::Sql Qt6::WebEngineQuick)
 add_test(NAME workspace COMMAND workspace_tests)
endif()
install(TARGETS arcreverse RUNTIME DESTINATION bin)
qt_generate_deploy_qml_app_script(TARGET arcreverse OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})

if(BUILD_TESTING)
 qt_add_executable(extension_probe tests/extension_probe.cpp)
 target_link_libraries(extension_probe PRIVATE Qt6::Widgets Qt6::WebEngineCore)
endif()
