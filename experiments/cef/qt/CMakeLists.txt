cmake_minimum_required(VERSION 3.28)
project(angle_cef_qt_spike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Quick)
if(NOT EXISTS "${CEF_WRAPPER_LIBRARY}")
 message(FATAL_ERROR "Build the standalone probe first and set CEF_WRAPPER_LIBRARY")
endif()
add_executable(angle_cef_qt_spike spike.cpp)
target_include_directories(angle_cef_qt_spike PRIVATE "${CEF_ROOT}")
target_link_libraries(angle_cef_qt_spike PRIVATE "${CEF_WRAPPER_LIBRARY}" "${CEF_ROOT}/Release/libcef.so" Qt6::Core Qt6::Gui Qt6::Quick X11 pthread dl)
set_target_properties(angle_cef_qt_spike PROPERTIES BUILD_RPATH "$ORIGIN")
file(GLOB runtime_files "${CEF_ROOT}/Release/*" "${CEF_ROOT}/Resources/*")
foreach(runtime IN LISTS runtime_files)
 get_filename_component(name "${runtime}" NAME)
 add_custom_command(TARGET angle_cef_qt_spike POST_BUILD COMMAND ${CMAKE_COMMAND} -E create_symlink "${runtime}" "$<TARGET_FILE_DIR:angle_cef_qt_spike>/${name}")
endforeach()
